Reference code — reveal after your attempt
def count_subarrays(a, k):
    counts = {0: 1}; total = answer = 0
    for x in a:
        total += x
        answer += counts.get(total-k, 0)
        counts[total] = counts.get(total, 0)+1
    return answer
