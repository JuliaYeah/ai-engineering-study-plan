def crawl(start, links):
    from urllib.parse import urlparse
    host = urlparse(start).hostname; seen = set(); todo = [start]
    while todo:
        url = todo.pop()
        if url in seen or urlparse(url).hostname != host: continue
        seen.add(url); todo.extend(links.get(url, []))
    return sorted(seen)
