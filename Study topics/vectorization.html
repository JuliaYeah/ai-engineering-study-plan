import numpy as np
def normalize_rows(x):
    x = np.asarray(x, dtype=float)
    if x.ndim != 2: raise ValueError("expected matrix")
    norms = np.linalg.norm(x, axis=1, keepdims=True)
    return np.divide(x, norms, out=np.zeros_like(x), where=norms != 0)
